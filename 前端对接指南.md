```javascript
// api/config.js
const BASE_URL = 'http://localhost:8080/bookkeeping-server'

// 获取Token
const getToken = () => wx.getStorageSync('token')

// 获取请求头
const getHeaders = () => ({
  'Content-Type': 'application/json',
```

### 1.2 统一请求方法

```javascript
// api/request.js

/***************************************
 * 统一请求方法
 * @param {string} url - 接口路径
 * @param {object} data - 请求参数
 * @param {string} method - 请求方法（默认POST）
 * @returns {Promise}
 ***************************************/
function request(url, data = {}, method = 'POST') {
  return new Promise((resolve, reject) => {
    wx.showLoading({ title: '加载中...' })
    
    wx.request({
      url: BASE_URL + url,
      method: method,
      data: data,
      header: getHeaders(),
      success: (res) => {
        wx.hideLoading()
        
        if (res.data.code === 200) {
          resolve(res.data.data)
        } else {
          wx.showToast({
            title: res.data.message || '请求失败',
            icon: 'none'
          })
          reject(res.data)
        }
      },
      fail: (err) => {
        wx.hideLoading()
        wx.showToast({
          title: '网络请求失败',
          icon: 'none'
        })
        reject(err)
      }
    })
  })
}

/***************************************
 * 快捷请求方法
 ***************************************/
const get = (url, data) => request(url, data, 'GET')
const post = (url, data) => request(url, data, 'POST')
const put = (url, data) => request(url, data, 'PUT')
const del = (url, data) => request(url, data, 'DELETE')

// 导出
module.exports = {
  request,
  get,
  post,
  put,
  del,
  BASE_URL
}
```

```javascript
// 用户模块
const userApi = {

   * 用户登出
```

```javascript
// pages/index/index.js
const userApi = require('../../api/user.js')
const recordApi = require('../../api/record.js')
const categoryApi = require('../../api/category.js')

Page({
  data: {
    userInfo: null,
    records: [],

      totalIncome: '0.00',
      totalExpense: '0.00',
      balance: '0.00'
    },

    this.loadCategories()
    this.loadStatistics()

   * 加载用户信息

    }
  },

   * 加载类别列表
   ***************************************/

      const expenseCategories = await categoryApi.getList(1)

        expenseCategories,
        incomeCategories
      })
    } catch (err) {
      console.error('获取类别失败', err)
    }
  },

   * 加载统计数据
   ***************************************/

      const now = new Date()
      const year = now.getFullYear()
      const month = String(now.getMonth() + 1).padStart(2, '0')
      const startDate = `${year}-${month}-01`
      const endDate = `${year}-${month}-31`
      
      const stats = await recordApi.getStatistics(startDate, endDate)
      this.setData({
        statistics: stats,
        currentMonth: `${year}年${month}月`
      })
    } catch (err) {
      console.error('获取统计失败', err)
    }
  },

  /***************************************
   * 加载账单列表
   ***************************************/
  async loadRecords(page = 1) {
    try {
      const result = await recordApi.getList({
        current: page,
        size: 10
      })
      
      if (page === 1) {
        this.setData({ records: result.records })
      } else {
        this.setData({
          records: [...this.data.records, ...result.records]
        })
      }
    } catch (err) {
      console.error('获取账单列表失败', err)
    }
  },

  /***************************************
   * 下拉刷新
   ***************************************/
  onPullDownRefresh() {
    Promise.all([
      this.loadUserInfo(),
      this.loadCategories(),
      this.loadStatistics(),
      this.loadRecords()
    ]).finally(() => {
      wx.stopPullDownRefresh()
    })
  },

  /***************************************
   * 上拉加载更多
   ***************************************/
  onReachBottom() {
```

```javascript
// utils/token.js

/***************************************
 * 保存Token
 ***************************************/
function saveToken(token, expiresIn) {
  wx.setStorageSync('token', token)
  wx.setStorageSync('tokenTime', Date.now())
  wx.setStorageSync('tokenExpire', expiresIn * 1000)
}

/***************************************
 * 获取Token
 ***************************************/
function getToken() {
  return wx.getStorageSync('token')
}

/***************************************
 * 检查Token是否过期
 ***************************************/
function isTokenExpired() {
  const tokenTime = wx.getStorageSync('tokenTime')
  const tokenExpire = wx.getStorageSync('tokenExpire')
  
  if (!tokenTime || !tokenExpire) {
    return true
  }
  
  return Date.now() - tokenTime > tokenExpire
}

/***************************************
 * 清除Token
 ***************************************/
function clearToken() {
  wx.removeStorageSync('token')
  wx.removeStorageSync('tokenTime')
  wx.removeStorageSync('tokenExpire')
}

/***************************************
 * 重新登录
 ***************************************/
async function reLogin() {
  clearToken()
  
  return new Promise((resolve, reject) => {
    wx.login({
      success: async (res) => {
        try {
          const data = await userApi.login(res.code)
          saveToken(data.token, data.expiresIn)
          resolve(data)
        } catch (err) {
          reject(err)
        }
      },
      fail: reject
    })
  })
}

// 导出
module.exports = {
  saveToken,
  getToken,
  isTokenExpired,
  clearToken,
  reLogin
}
```

```javascript
// utils/error.js

/***************************************
 * 全局错误处理
 ***************************************/
function handleError(err, showToast = true) {
  console.error('请求错误:', err)
  
  if (err.code === 401) {
    // Token过期，跳转登录
    wx.redirectTo({
      url: '/pages/login/login'
    })
    return
  }
  
  if (err.code === 403) {
    wx.showToast({
      title: '没有权限',
      icon: 'none'
    })
    return
  }
  
  if (err.code === 500) {
    wx.showToast({
      title: '服务器错误',
      icon: 'none'
    })
    return
  }
  
  if (showToast && err.message) {
    wx.showToast({
      title: err.message,
      icon: 'none'
    })
  }
}

// 导出
module.exports = {
  handleError
}
```
